months = d3.range(24).map(i => new Date(2023, i, 1))
series = ["Bus", "Light rail", "Commuter rail", "Bike share", "Vanpool"]
base = [1820, 610, 215, 38, 24]
rows = series.flatMap((s, k) => months.map((m, i) => ({
  month: m, mode: s,
  value: Math.round(base[k] * (1 + 0.012 * i) * (1 + 0.05 * Math.sin((i % 12 - 2) / 12 * 2 * Math.PI)))
})))
chart = n => getComputedStyle(document.documentElement).getPropertyValue(`--wfrc-chart-${n}`).trim()
